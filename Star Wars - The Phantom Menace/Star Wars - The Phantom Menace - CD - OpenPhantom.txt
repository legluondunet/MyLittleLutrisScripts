files:
- game_cd: https://archive.org/download/phantom-menace-pc-game/THEPHANTOMMENACE.ISO
- game_mod: https://github.com/OpenPhantom/OpenPhantom/releases/download/i1.3/OpenPhantom_Installer.exe
game:
  exe: drive_c/Program Files (x86)/LucasArts/The Phantom Menace/WMAIN.EXE
  prefix: $GAMEDIR
installer:
- task:
    name: create_prefix
    prefix: $GAMEDIR
- extract:
    dst: $GAMEDIR/drive_c/CD
    file: game_cd
    format: iso
- task:
    description: Installing game...You can use your own CD or you will find gaem files
      at "c:\CD"
    executable: game_mod
    name: wineexec
    prefix: $GAMEDIR
- execute:
    command: cd "$GAMEDIR/drive_c/Program Files (x86)/LucasArts/The Phantom Menace"
      ; sed -i s/MenuKeepsResolution=.*/MenuKeepsResolution=0/ engine_fixes.ini ;
- move:
    dst: $CACHE
    src: $GAMEDIR/drive_c/CD
wine:
  overrides:
    ddraw: n,b
    dinput: n,b
