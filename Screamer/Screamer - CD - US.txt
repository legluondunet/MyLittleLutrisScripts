files:
- game_files: https://archive.org/download/screamer-pc-redump/SCREAMER-PC-REDUMP/Screamer%20%28USA%29.zip
game:
  main_file: $GAMEDIR/dosbox.conf
installer:
- extract:
    dst: $GAMEDIR/D
    file: game_files
- execute:
    command: mkdir $GAMEDIR/C
- write_file:
    content: '[cpu]

      core=dynamic

      cycles=150000

      cycleup=500

      cycledown=20

      [autoexec]

      @echo off

      mount c C

      imgmount d "D\Screamer (USA).cue" -t iso -fs iso

      :start

      if not exist "c:\screamer\screamer.bat" goto :install

      :play

      c:\

      cd c:\screamer

      call screamer.bat

      goto :exit

      :install

      d:\

      call d:\install.bat

      goto :start

      :exit

      exit'
    file: $GAMEDIR/dosbox.conf
