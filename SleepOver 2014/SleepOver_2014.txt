files:
- game_sdk: https://www.renpy.org/dl/8.5.3/renpy-8.5.3-sdk.tar.bz2
game:
  args: sleepover
  exe: renpy.sh
installer:
- extract:
    dst: $GAMEDIR
    file: $game_sdk
- chmodx: $GAMEDIR/renpy.sh
- input_menu:
    description: 'Choose the game''s language:'
    id: LANG
    options:
    - en: English
    - fr: French
    preselect: en
- execute:
    command: cd "$CACHE" ; if [ "$INPUT_LANG" = "en" ]; then wget -O sleepover.zip
      --no-check-certificate 'https://archive.org/download/sleep-over_202606/sleepover_english.zip'
      ;  elif [ "$INPUT_LANG" = "fr" ]; then wget -O sleepover.zip --no-check-certificate
      'https://archive.org/download/sleep-over_202606/sleepover_french.zip' ; fi ;
- extract:
    dst: $CACHE/tmp
    file: $CACHE/sleepover.zip
- execute:
    command: cd "$CACHE"; sleepodir=$(dirname "$(find "$CACHE/tmp" -type f |grep -i
      archive.rpa)"); echo "$sleepodir" ; cp -r "$sleepodir" sleepover;
- merge:
    dst: $GAMEDIR/sleepover
    src: $CACHE/sleepover
